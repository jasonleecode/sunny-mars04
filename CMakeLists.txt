cmake_minimum_required(VERSION 3.16)
project(sunny_mars04 VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
include(CMakePackageConfigHelpers)
option(MARS04_BUILD_TOOLS "Build capture CLI (requires libpng)" ON)
option(MARS04_BUILD_GUI "Build the SDL2/Cairo live camera viewer" OFF)
option(MARS04_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
find_package(Threads REQUIRED)
add_library(mars04 src/camera.cpp src/protocol.cpp)
add_library(mars04::mars04 ALIAS mars04)
target_compile_features(mars04 PUBLIC cxx_std_17)
target_include_directories(mars04 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(mars04 PRIVATE PkgConfig::LIBUSB Threads::Threads)
set_target_properties(mars04 PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(mars04 PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  if(MARS04_SANITIZERS)
    target_compile_options(mars04 PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mars04 PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(MARS04_BUILD_TOOLS)
  find_package(PNG REQUIRED)
  add_executable(mars04-capture src/main.cpp)
  target_link_libraries(mars04-capture PRIVATE mars04 PNG::PNG Threads::Threads)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(mars04-capture PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  endif()
  install(TARGETS mars04-capture RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(MARS04_BUILD_GUI)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo)
  add_executable(mars04-viewer src/viewer.cpp)
  target_link_libraries(mars04-viewer PRIVATE mars04 PkgConfig::SDL2 PkgConfig::CAIRO Threads::Threads)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(mars04-viewer PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  endif()
  install(TARGETS mars04-viewer RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES desktop/mars04-viewer.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
endif()
if(BUILD_TESTING)
  add_executable(mars04-tests tests/protocol_test.cpp)
  target_link_libraries(mars04-tests PRIVATE mars04)
  add_test(NAME protocol COMMAND mars04-tests)
endif()
install(TARGETS mars04 EXPORT mars04Targets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES udev/70-sunny-mars04.rules DESTINATION ${CMAKE_INSTALL_DATADIR}/sunny-mars04)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES docs/protocol.md docs/validation.md DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)
configure_package_config_file(cmake/mars04Config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/mars04Config.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/mars04)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/mars04ConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT mars04Targets NAMESPACE mars04:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/mars04)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/mars04Config.cmake ${CMAKE_CURRENT_BINARY_DIR}/mars04ConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/mars04)
